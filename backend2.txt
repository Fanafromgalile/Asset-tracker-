const MAX_HISTORY_ROWS = 5000;

function getTargetSpreadsheet() {
  let raw = PropertiesService.getScriptProperties().getProperty("SHEET_ID");
  if (!raw || !raw.trim()) {
    throw new Error("SHEET_ID is empty. Add it under Project Settings > Script Properties.");
  }
  raw = raw.trim();
  const urlMatch = raw.match(/\/spreadsheets\/d\/([a-zA-Z0-9-_]+)/);
  const cleanId = urlMatch ? urlMatch[1] : raw.replace(/[^a-zA-Z0-9-_]/g, "");
  return SpreadsheetApp.openById(cleanId);
}

function sanitizeCell(val) {
  if (val === null || val === undefined) return "";
  if (typeof val === "number" || typeof val === "boolean" || val instanceof Date) return val;
  let str = String(val).replace(/[\r\n\t]+/g, " ").trim();
  if (/^[=+\-@]/.test(str)) {
    str = "'" + str;
  }
  return str;
}

function sha256Hex(input) {
  const rawBytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, input, Utilities.Charset.UTF_8);
  return rawBytes.map(b => {
    const v = b < 0 ? b + 256 : b;
    return ("0" + v.toString(16)).slice(-2);
  }).join("");
}

function getSettingsMap(ss) {
  const sheet = ss.getSheetByName("Settings");
  if (!sheet || sheet.getLastRow() < 2) return {};
  const data = sheet.getRange(2, 1, sheet.getLastRow() - 1, 2).getValues();
  const map = {};
  for (let i = 0; i < data.length; i++) {
    const k = String(data[i][0] || "").trim();
    if (k) map[k] = data[i][1];
  }
  return map;
}

function getDistanceMeters(lat1, lon1, lat2, lon2) {
  const R = 6371000;
  const dLat = (lat2 - lat1) * Math.PI / 180;
  const dLon = (lon2 - lon1) * Math.PI / 180;
  const a = Math.sin(dLat / 2) ** 2 + Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * Math.sin(dLon / 2) ** 2;
  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
}

function doGet(e) {
  let dbConnected = false, sheetName = "", dbError = "";
  try {
    const ss = getTargetSpreadsheet();
    sheetName = ss.getName();
    dbConnected = true;
  } catch (err) {
    dbError = err.message;
  }

  const execUrl = (ScriptApp.getService().getUrl() || "").replace(/\/dev$/, "/exec");
  const tpl = HtmlService.createTemplateFromFile("Enrol");
  tpl.bootJson = JSON.stringify({ dbConnected, sheetName, dbError, execUrl }).replace(/</g, "\\u003c");

  return tpl.evaluate()
    .setTitle("Laptop Enrollment Portal")
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

/* ============================================================================
   SHARED USER & ADMIN ACCOUNT AUTHENTICATION
   ============================================================================ */

function registerPortalAccount(email, fullName, password) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);

  try {
    const cleanEmail = String(email || "").trim().toLowerCase();
    const cleanName = sanitizeCell(fullName || cleanEmail.split("@")[0]);
    const cleanPass = String(password || "");

    if (!cleanEmail || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(cleanEmail)) {
      return { success: false, message: "Enter a valid email address." };
    }
    if (cleanPass.length < 6) {
      return { success: false, message: "Password must be at least 6 characters." };
    }

    const ss = getTargetSpreadsheet();
    const usersSheet = ss.getSheetByName("Users");
    if (!usersSheet) {
      return { success: false, message: "Users tab missing. Run setupSheets() in Project A first." };
    }

    const lastRow = usersSheet.getLastRow();
    let hasAnyAdmin = false;
    if (lastRow >= 2) {
      const existing = usersSheet.getRange(2, 1, lastRow - 1, 3).getValues();
      for (let i = 0; i < existing.length; i++) {
        if (String(existing[i][0]).trim().toLowerCase() === cleanEmail) {
          return { success: false, message: "Account already exists. Please sign in." };
        }
        if (String(existing[i][2]).trim() === "Admin") hasAnyAdmin = true;
      }
    }

    if (hasAnyAdmin) {
      const sMap = getSettingsMap(ss);
      if (String(sMap["Allow Self Signup"] || "ON").toUpperCase() === "OFF") {
        return { success: false, message: "Self-registration is currently disabled by the Administrator." };
      }
      const allowedRaw = String(sMap["Allowed Domains"] || "").trim();
      if (allowedRaw) {
        const domains = allowedRaw.split(",").map(d => d.trim().toLowerCase()).filter(Boolean);
        const userDom = cleanEmail.split("@")[1] || "";
        if (domains.length && !domains.includes(userDom)) {
          return { success: false, message: "Email domain is not authorized for registration." };
        }
      }
    }

    const role = hasAnyAdmin ? "User" : "Admin";
    const salt = Utilities.getUuid().replace(/-/g, "");
    const hash = sha256Hex(salt + ":" + cleanPass);
    const now = new Date();

    usersSheet.appendRow([cleanEmail, cleanName, role, salt, hash, now, now]);

    const userObj = { email: cleanEmail, name: cleanName, role: role };
    const token = issuePortalSession(userObj);
    return { success: true, sessionToken: token, user: userObj };
  } catch (err) {
    return { success: false, message: err.message };
  } finally {
    lock.releaseLock();
  }
}

function loginPortalAccount(email, password) {
  const cleanEmail = String(email || "").trim().toLowerCase();
  const cleanPass = String(password || "");

  const ss = getTargetSpreadsheet();
  const usersSheet = ss.getSheetByName("Users");
  if (!usersSheet || usersSheet.getLastRow() < 2) {
    return { success: false, message: "No accounts registered yet. Create an account first." };
  }

  const data = usersSheet.getRange(2, 1, usersSheet.getLastRow() - 1, 5).getValues();
  for (let i = 0; i < data.length; i++) {
    if (String(data[i][0]).trim().toLowerCase() === cleanEmail) {
      const fullName = String(data[i][1] || cleanEmail);
      const role = String(data[i][2] || "User");
      const salt = String(data[i][3] || "");
      const storedHash = String(data[i][4] || "");

      if (role === "Disabled") return { success: false, message: "Account disabled by Administrator." };
      if (sha256Hex(salt + ":" + cleanPass) !== storedHash) return { success: false, message: "Incorrect password." };

      usersSheet.getRange(i + 2, 7).setValue(new Date());
      const userObj = { email: cleanEmail, name: fullName, role: role };
      const token = issuePortalSession(userObj);
      return { success: true, sessionToken: token, user: userObj };
    }
  }
  return { success: false, message: "Account not found. Please create an account." };
}

function issuePortalSession(userObj) {
  const token = Utilities.getUuid().replace(/-/g, "") + Utilities.getUuid().replace(/-/g, "");
  const record = { email: userObj.email, name: userObj.name, role: userObj.role, exp: Date.now() + 86400000 };
  CacheService.getScriptCache().put("psess_" + token, JSON.stringify(record), 21600);
  PropertiesService.getScriptProperties().setProperty("PSESS_" + sha256Hex(token).slice(0, 24), JSON.stringify(record));
  return token;
}

function verifyPortalSession(sessionToken) {
  const token = String(sessionToken || "").trim();
  if (!token) throw new Error("AUTH_REQUIRED");
  const cached = CacheService.getScriptCache().get("psess_" + token);
  let user = cached ? JSON.parse(cached) : null;
  if (!user) {
    const raw = PropertiesService.getScriptProperties().getProperty("PSESS_" + sha256Hex(token).slice(0, 24));
    if (raw) user = JSON.parse(raw);
  }
  if (!user || Date.now() > user.exp) throw new Error("AUTH_REQUIRED");
  return user;
}

function getMyEnrolledDevices(sessionToken) {
  const user = verifyPortalSession(sessionToken);
  const ss = getTargetSpreadsheet();
  const assetsSheet = ss.getSheetByName("Assets");
  const mine = [];
  if (assetsSheet && assetsSheet.getLastRow() >= 2) {
    const data = assetsSheet.getRange(2, 1, assetsSheet.getLastRow() - 1, 17).getValues();
    for (let i = 0; i < data.length; i++) {
      if (String(data[i][1]).trim().toLowerCase() === user.email || String(data[i][15]).trim().toLowerCase() === user.email) {
        mine.push({
          id: String(data[i][0] || ""),
          status: String(data[i][2] || ""),
          place: String(data[i][5] || ""),
          updated: data[i][6] instanceof Date ? data[i][6].toISOString() : String(data[i][6] || ""),
          tokenStatus: String(data[i][16] || "Active")
        });
      }
    }
  }
  return { user: user, devices: mine };
}

function enrollLaptopAuthenticated(payload) {
  const user = verifyPortalSession(payload.sessionToken);
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);

  try {
    const assetId = sanitizeCell(payload.assetId || "");
    const model = sanitizeCell(payload.model || "");
    const lat = parseFloat(payload.lat);
    const lng = parseFloat(payload.lng);
    const placeName = sanitizeCell(payload.placeName || "");
    const method = sanitizeCell(payload.method || "Portal Check-In");
    const accuracy = payload.accuracy !== undefined && payload.accuracy !== "" ? Number(payload.accuracy) : 50;

    if (!assetId) return { success: false, message: "Laptop / Asset ID is required." };

    const rawDeviceToken = Utilities.getUuid().replace(/-/g, "") + Utilities.getUuid().replace(/-/g, "");
    const tokenHash = sha256Hex(rawDeviceToken);
    const timestamp = new Date();

    const ss = getTargetSpreadsheet();
    const devSheet = ss.getSheetByName("Authorized_Devices");
    const assetsSheet = ss.getSheetByName("Assets");
    const historySheet = ss.getSheetByName("History");

    const devData = devSheet.getDataRange().getValues();
    let devFound = false;
    for (let i = 1; i < devData.length; i++) {
      if (String(devData[i][0]).trim().toLowerCase() === assetId.toLowerCase()) {
        devSheet.getRange(i + 1, 2, 1, 5).setValues([[tokenHash, user.email, "Active", timestamp, timestamp]]);
        devFound = true;
        break;
      }
    }
    if (!devFound) {
      devSheet.appendRow([assetId, tokenHash, user.email, "Active", timestamp, timestamp]);
    }

    const hasCoords = !isNaN(lat) && !isNaN(lng) && !(lat === 0 && lng === 0);
    if (hasCoords) {
      historySheet.appendRow([timestamp, assetId, lat, lng, placeName, method, accuracy, "", "", ""]);
    }

    const assetData = assetsSheet.getDataRange().getValues();
    let assetFound = false;
    for (let i = 1; i < assetData.length; i++) {
      if (String(assetData[i][0]).trim().toLowerCase() === assetId.toLowerCase()) {
        assetsSheet.getRange(i + 1, 2, 1, 8).setValues([[
          user.name || user.email,
          assetData[i][2] || "Active",
          hasCoords ? lat : assetData[i][3],
          hasCoords ? lng : assetData[i][4],
          placeName || assetData[i][5] || "",
          timestamp,
          hasCoords ? method : (assetData[i][7] || "Enrolled"),
          hasCoords ? accuracy : (assetData[i][8] || "")
        ]]);
        if (model) assetsSheet.getRange(i + 1, 12).setValue(model);
        assetsSheet.getRange(i + 1, 16, 1, 2).setValues([[user.email, "Active"]]);
        assetFound = true;
        break;
      }
    }

    if (!assetFound) {
      assetsSheet.appendRow([
        assetId, user.name || user.email, "Active",
        hasCoords ? lat : "", hasCoords ? lng : "", placeName,
        timestamp, hasCoords ? method : "Enrolled", hasCoords ? accuracy : "",
        "", "", model, "", "", "", user.email, "Active"
      ]);
    }

    const execUrl = (ScriptApp.getService().getUrl() || "").replace(/\/dev$/, "/exec");

    return {
      success: true,
      assetId: assetId,
      email: user.email,
      deviceToken: rawDeviceToken,
      webhookUrl: execUrl
    };
  } catch (err) {
    return { success: false, message: err.message };
  } finally {
    lock.releaseLock();
  }
}

/* ============================================================================
   REAL-TIME STOLEN & GEOFENCE BREACH/RETURN ALERTS
   ============================================================================ */

function evaluateSecurityAlerts(ss, assetId, user, status, prevLat, prevLng, newLat, newLng, placeName, ssid, battery) {
  try {
    const sMap = getSettingsMap(ss);
    const recipients = String(sMap["Alert Emails"] || "").trim();
    if (!recipients) return;

    const cache = CacheService.getScriptCache();
    const alertKey = "alert_cool_" + assetId.toLowerCase();
    if (cache.get(alertKey)) return;

    const mapsLink = `https://www.google.com/maps?q=${newLat},${newLng}`;
    const osmLink = `https://www.openstreetmap.org/?mlat=${newLat}&mlon=${newLng}#map=17/${newLat}/${newLng}`;
    const battText = battery !== "" ? ` &bull; Battery: ${battery}%` : "";

    // 1. Stolen Device Ping Alert
    if (status === "Stolen") {
      cache.put(alertKey, "1", 1800); // 30-min cooldown for stolen beacons
      MailApp.sendEmail({
        to: recipients,
        subject: `URGENT: Stolen Laptop Detected - ${assetId} (${placeName || newLat + "," + newLng})`,
        htmlBody: `
          <div style="font-family:Arial,sans-serif;max-width:500px;padding:20px;border:2px solid #dc2626;border-radius:10px;">
            <h2 style="margin:0 0 10px;color:#dc2626;">Stolen Laptop Check-In Detected</h2>
            <p><b>Asset ID:</b> ${assetId}<br>
            <b>Assigned User:</b> ${user}<br>
            <b>Location:</b> ${placeName || "Unknown"} (${newLat}, ${newLng})<br>
            <b>Wi-Fi SSID:</b> ${ssid || "N/A"}${battText}</p>
            <p><a href="${mapsLink}">Open in Google Maps</a> | <a href="${osmLink}">Open in OpenStreetMap</a></p>
          </div>
        `
      });
      return;
    }

    // 2. Geofence Exit / Return Alerts
    if (String(sMap["Geofence Alerts"] || "ON").toUpperCase() !== "ON") return;
    if (isNaN(prevLat) || isNaN(prevLng)) return;

    const officesSheet = ss.getSheetByName("Offices");
    if (!officesSheet || officesSheet.getLastRow() < 2) return;
    const oData = officesSheet.getRange(2, 1, officesSheet.getLastRow() - 1, 5).getValues();

    let prevBranch = null;
    let currentBranch = null;

    for (let i = 0; i < oData.length; i++) {
      const bLat = parseFloat(oData[i][2]);
      const bLng = parseFloat(oData[i][3]);
      const bRad = Number(oData[i][4]) || 100;
      if (isNaN(bLat) || isNaN(bLng)) continue;

      if (getDistanceMeters(prevLat, prevLng, bLat, bLng) <= bRad) prevBranch = String(oData[i][1]);
      if (getDistanceMeters(newLat, newLng, bLat, bLng) <= bRad) currentBranch = String(oData[i][1]);
    }

    if (prevBranch && !currentBranch) {
      cache.put(alertKey, "1", 3600);
      MailApp.sendEmail({
        to: recipients,
        subject: `Geofence Exit Alert: ${assetId} left ${prevBranch}`,
        htmlBody: `
          <div style="font-family:Arial,sans-serif;max-width:500px;padding:20px;border:1px solid #f59e0b;border-radius:10px;">
            <h2 style="margin:0 0 10px;color:#b45309;">Laptop Left Office Geofence</h2>
            <p><b>Asset ID:</b> ${assetId} (${user})<br>
            <b>Departed Branch:</b> ${prevBranch}<br>
            <b>Current Location:</b> ${placeName || "Off-site"} (${newLat}, ${newLng})<br>
            <b>Wi-Fi SSID:</b> ${ssid || "N/A"}${battText}</p>
            <p><a href="${mapsLink}">View Location on Map</a></p>
          </div>
        `
      });
    } else if (!prevBranch && currentBranch) {
      cache.put(alertKey, "1", 3600);
      MailApp.sendEmail({
        to: recipients,
        subject: `Office Return Notice: ${assetId} arrived at ${currentBranch}`,
        htmlBody: `
          <div style="font-family:Arial,sans-serif;max-width:500px;padding:20px;border:1px solid #16a34a;border-radius:10px;">
            <h2 style="margin:0 0 10px;color:#15803d;">Laptop Returned to Office Geofence</h2>
            <p><b>Asset ID:</b> ${assetId} (${user})<br>
            <b>Arrived at Branch:</b> ${currentBranch}<br>
            <b>Wi-Fi SSID:</b> ${ssid || "N/A"}${battText}</p>
          </div>
        `
      });
    }
  } catch (e) {}
}

/* ============================================================================
   DUAL-MODE doPost: PORTAL RPCs & HARDENED WINDOWS TELEMETRY INGESTION
   ============================================================================ */

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse({ status: "error", message: "Empty request body." });
    }

    const payload = JSON.parse(e.postData.contents);

    if (payload.rpcAction) {
      const act = payload.rpcAction;
      let res;
      if (act === "registerPortalAccount") res = registerPortalAccount(payload.email, payload.fullName, payload.password);
      else if (act === "loginPortalAccount") res = loginPortalAccount(payload.email, payload.password);
      else if (act === "getMyEnrolledDevices") res = getMyEnrolledDevices(payload.sessionToken);
      else if (act === "enrollLaptopAuthenticated") res = enrollLaptopAuthenticated(payload);
      else res = { error: "Unknown RPC action" };
      return jsonResponse({ ok: true, data: res });
    }

    const lock = LockService.getScriptLock();
    lock.waitLock(10000);
    try {
      const assetId = sanitizeCell(payload.id || "");
      const rawToken = String(payload.token || "").trim();
      if (!assetId || !rawToken) return jsonResponse({ status: "error", message: "Missing device id or token." });

      const ss = getTargetSpreadsheet();
      const devSheet = ss.getSheetByName("Authorized_Devices");
      if (!devSheet || devSheet.getLastRow() < 2) return jsonResponse({ status: "error", message: "Unauthorized device." });

      const incomingHash = sha256Hex(rawToken);
      const devData = devSheet.getDataRange().getValues();
      let authorizedRow = -1, enrolledEmail = "";

      for (let i = 1; i < devData.length; i++) {
        if (String(devData[i][0]).trim().toLowerCase() === assetId.toLowerCase()) {
          if (String(devData[i][3] || "").trim() !== "Active") return jsonResponse({ status: "error", message: "Device token revoked." });
          if (String(devData[i][1] || "").trim() === incomingHash) {
            authorizedRow = i + 1;
            enrolledEmail = String(devData[i][2] || "");
          }
          break;
        }
      }

      if (authorizedRow === -1) return jsonResponse({ status: "error", message: "Unauthorized device token." });

      let lat = parseFloat(payload.lat);
      let lng = parseFloat(payload.lng);
      if (isNaN(lat) || isNaN(lng) || lat < -90 || lat > 90 || lng < -180 || lng > 180 || (lat === 0 && lng === 0)) {
        return jsonResponse({ status: "error", message: "Invalid GPS coordinates." });
      }

      let placeName = sanitizeCell(payload.placeName || "");
      let method = sanitizeCell(payload.method || "IP Geo");
      let accuracy = payload.accuracy !== undefined && payload.accuracy !== "" ? Number(payload.accuracy) : 5000;
      const ip = sanitizeCell(payload.ip || "");
      const serial = sanitizeCell(payload.serial || "");
      const model = sanitizeCell(payload.model || "");
      const ssid = sanitizeCell(payload.ssid || "");
      const battery = payload.battery !== undefined && payload.battery !== "" ? Number(payload.battery) : "";
      const winUser = sanitizeCell(payload.winUser || "");
      const timestamp = new Date();

      const assetsSheet = ss.getSheetByName("Assets");
      const historySheet = ss.getSheetByName("History");
      const data = assetsSheet.getDataRange().getValues();

      let currentStatus = "Active";
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]).trim().toLowerCase() === assetId.toLowerCase()) {
          currentStatus = String(data[i][2] || "Active").trim();
          const prevTime = data[i][6] instanceof Date ? data[i][6] : new Date(data[i][6]);
          const secSinceLast = !isNaN(prevTime.getTime()) ? (timestamp - prevTime) / 1000 : 999;

          // Anti-flood rate limit: skip duplicate pings within 30s unless device is Stolen
          if (secSinceLast < 30 && currentStatus !== "Stolen") {
            return jsonResponse({ status: "ok", message: "Throttled duplicate ping.", lostMode: false });
          }

          const currentUser = String(data[i][1] || "").trim();
          const prevLat = parseFloat(data[i][3]);
          const prevLng = parseFloat(data[i][4]);
          const prevPlace = String(data[i][5] || "");
          const prevAcc = Number(data[i][8]) || 99999;
          const prevIp = String(data[i][9] || "");

          const hoursSinceLast = secSinceLast / 3600;
          if (ip !== "" && ip === prevIp && accuracy >= 1000 && prevAcc <= 150 && hoursSinceLast < 6 && !isNaN(prevLat)) {
            lat = prevLat; lng = prevLng; placeName = prevPlace || placeName; accuracy = prevAcc;
            method = sanitizeCell(String(data[i][7] || "Windows Wi-Fi/GPS") + " (Retained)");
          }

          const resolvedUser = (currentUser === "" || currentUser === "Unassigned") ? (enrolledEmail || winUser || "Unassigned") : data[i][1];

          assetsSheet.getRange(i + 1, 2, 1, 16).setValues([[
            resolvedUser, currentStatus, lat, lng, placeName || prevPlace, timestamp,
            method, accuracy, ip, serial || data[i][10] || "", model || data[i][11] || "",
            ssid, battery, winUser || data[i][14] || "", enrolledEmail || data[i][15] || "", "Active"
          ]]);

          evaluateSecurityAlerts(ss, assetId, resolvedUser, currentStatus, prevLat, prevLng, lat, lng, placeName || prevPlace, ssid, battery);
          break;
        }
      }

      devSheet.getRange(authorizedRow, 6).setValue(timestamp);
      historySheet.appendRow([timestamp, assetId, lat, lng, placeName, method, accuracy, ip, ssid, battery]);
      const histRows = historySheet.getLastRow();
      if (histRows > MAX_HISTORY_ROWS + 200) historySheet.deleteRows(2, histRows - MAX_HISTORY_ROWS);

      return jsonResponse({
        status: "ok",
        message: "Telemetry logged.",
        lostMode: currentStatus === "Stolen"
      });
    } finally {
      lock.releaseLock();
    }
  } catch (err) {
    return jsonResponse({ status: "error", message: err.message });
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
